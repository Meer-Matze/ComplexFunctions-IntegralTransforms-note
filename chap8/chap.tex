\chapter{Laplace变换}

\section{Laplace变换的定义与存在性}

\subsection{定义}

\begin{definition}[Laplace 变换]
设 $f(t)$ 是定义在 $[0,+\infty)$ 上的实值（或复值）函数．若积分
\[
F(s) = \L[f](s) \defeq \int_0^{+\infty} f(t)\e^{-st}\,\d t
\]
对参数 $s$（复数）收敛，则称 $F(s)$ 为 $f(t)$ 的\textbf{Laplace 变换}，记作 $F(s) = \L[f(t)]$ 或 $F(s) = \mathscr{L}\{f(t)\}$．
\end{definition}

\begin{definition}[Laplace 逆变换]
设 $F(s)$ 为某函数 $f(t)$ 的 Laplace 变换．称
\[
f(t) = \L^{-1}[F](t) \defeq \frac{1}{2\pi i}\int_{\sigma-i\infty}^{\sigma+i\infty} F(s)\e^{st}\,\d s
\]
为 $F(s)$ 的\textbf{Laplace 逆变换}，其中积分沿直线 $\Re s = \sigma$ 实施，$\sigma$ 须大于 $F(s)$ 所有奇点的实部．
\end{definition}

\subsection{存在性条件}

\begin{definition}[指数阶函数]
设 $f$ 在 $[0,+\infty)$ 上分段连续．若存在常数 $M\ge 0$ 与 $\alpha\in\R$，使得
\[
\abs{f(t)} \le M\e^{\alpha t},\quad \forall t\ge 0,
\]
则称 $f$ 为\textbf{指数阶函数}，并称 $\alpha$ 为其\textbf{增长指数}．
\end{definition}

\begin{theorem}[Laplace 变换存在性定理]
设 $f$ 在 $[0,+\infty)$ 上分段连续，且为指数阶函数．则
\begin{enumerate}
    \item $\L[f](s)$ 在半平面 $\Re s > \alpha$ 内存在且解析；
    \item 当 $\Re s\to+\infty$ 时 $F(s)\to 0$．
\end{enumerate}
\end{theorem}

\begin{theorem}[唯一性定理]
设 $f_1,f_2$ 在 $[0,+\infty)$ 上分段连续，且 $F_1(s)=F_2(s)$ 在某半平面内成立．则 $f_1(t) = f_2(t)$ 在 $f_1,f_2$ 的所有连续点处成立．
\end{theorem}

\begin{example}
    求下列函数的 Laplace 变换：
    \begin{tasks}[label=(\arabic*)](2)
        \task $\displaystyle f(t) = \begin{cases}
            3, & 0\le t<2\\
            -1, & 2\le t<4\\
            0, & t\ge 4
            \end{cases}$
        \task $\displaystyle f(t) = \cos^2 t$
        \task $\displaystyle f(t) = \delta(t)\cos t - u(t)\sin t$
    \end{tasks} 
\end{example}
\begin{solution}
    \begin{enumerate}[label=(\arabic*)]
        \item 直接按定义积分：
        \[
        \begin{aligned}
        F(s) &= \int_0^{2} 3\e^{-st}\,\d t + \int_2^{4} (-1)\e^{-st}\,\d t + \int_4^{+\infty} 0\cdot\e^{-st}\,\d t \\[1em]
        &= 3\left[-\frac{\e^{-st}}{s}\right]_0^2 - \left[-\frac{\e^{-st}}{s}\right]_2^4 \\[1em]
        &= \frac{3(1-\e^{-2s})}{s} + \frac{\e^{-4s}-\e^{-2s}}{s}
        = \frac{3 - 4\e^{-2s} + \e^{-4s}}{s},\qquad \Re s>0.
        \end{aligned}
        \]

        \item 利用三角恒等式 $\cos^2 t=\dfrac{1+\cos 2t}{2}$，并引用基本变换对：
        \[
        F(s) = \L\!\left[\frac{1}{2}\right] + \L\!\left[\frac{1}{2}\cos 2t\right]
        = \frac{1}{2s} + \frac{1}{2}\cdot\frac{s}{s^2+4}
        = \frac{s^2+2}{s(s^2+4)},\qquad \Re s>0.
        \]

        \item 由 $\L[\delta(t)]=1$ 与筛选性质 $\L[\delta(t)\cos t]=\cos 0=1$，又 $\L[u(t)\sin t]=\L[\sin t]=\dfrac{1}{s^2+1}$，故
        \[
        F(s) = 1 - \frac{1}{s^2+1} = \frac{s^2}{s^2+1},\qquad \Re s>0.
        \]
    \end{enumerate}
\end{solution}

\section{Laplace变换的性质}

\begin{theorem}[线性性质]
设 $\L[f_1]=F_1(s)$，$\L[f_2]=F_2(s)$，$\alpha,\beta\in\C$．则
\[
\L[\alpha f_1 + \beta f_2] = \alpha F_1(s) + \beta F_2(s).
\]
\end{theorem}

\begin{theorem}[位移性质 / $s$ 域平移]
\[
\L[\e^{at}f(t)] = F(s-a),\quad \Re s > \Re a + \alpha.
\]
\end{theorem}

\begin{theorem}[延迟性质 / $t$ 域平移]
设 $t_0>0$，$f$ 在 $t<t_0$ 时补零．则
\[
\L[f(t-t_0)u(t-t_0)] = \e^{-st_0}F(s).
\]
\end{theorem}

\begin{theorem}[相似性质]
对 $a>0$，
\[
\L[f(at)] = \frac{1}{a}F\!\left(\frac{s}{a}\right).
\]
\end{theorem}

\begin{theorem}[微分性质：$t$ 域]
设 $f$ 连续，$f'$ 分段连续且指数阶．则
\[
\L[f'(t)] = sF(s) - f(0),\qquad
\L[f^{(n)}(t)] = s^n F(s) - s^{n-1}f(0) - \cdots - f^{(n-1)}(0).
\]
\end{theorem}

\begin{theorem}[微分性质：$s$ 域]
设 $tf(t)$ 指数阶．则
\[
\L[-tf(t)] = F'(s),\qquad
\L[(-t)^n f(t)] = F^{(n)}(s).
\]
\end{theorem}

\begin{theorem}[积分性质]
\[
\L\!\left[\int_0^t f(\tau)\,\d\tau\right] = \frac{1}{s}F(s),\qquad
\L\!\left[\frac{f(t)}{t}\right] = \int_s^{+\infty} F(\sigma)\,\d\sigma.
\]
\end{theorem}

\begin{theorem}[初值与终值定理]
设 $f,f'$ 均存在 Laplace 变换，且 $\lim_{s\to\infty} sF(s)$ 存在．
\begin{enumerate}
    \item 初值定理：$\displaystyle\lim_{t\to 0^+} f(t) = \lim_{s\to\infty} sF(s)$；
    \item 终值定理（$f$ 的所有奇点位于 $\Re s<0$）：$\displaystyle\lim_{t\to+\infty} f(t) = \lim_{s\to 0} sF(s)$．
\end{enumerate}
\end{theorem}

\begin{theorem}[卷积定理]
设 $\L[f]=F(s)$，$\L[g]=G(s)$，Laplace 卷积
\[
(f*g)(t) \defeq \int_0^t f(\tau)g(t-\tau)\,\d\tau,\quad t\ge 0.
\]
则
\[
\L[f*g] = F(s)G(s),\qquad \L^{-1}[F(s)G(s)] = (f*g)(t).
\]
\end{theorem}

\section{常用函数的Laplace变换}

\begin{theorem}[基本函数的 Laplace 变换]
\[
\begin{array}{lll}
\textbf{(1)} & \L[1] = \dfrac{1}{s}, & \Re s > 0 \\[6pt]
\textbf{(2)} & \L[t^n] = \dfrac{n!}{s^{n+1}}, & \Re s > 0,\; n\in\N \\[6pt]
\textbf{(3)} & \L[\e^{at}] = \dfrac{1}{s-a}, & \Re s > \Re a \\[6pt]
\textbf{(4)} & \L[\sin\omega t] = \dfrac{\omega}{s^2+\omega^2}, & \Re s > 0 \\[6pt]
\textbf{(5)} & \L[\cos\omega t] = \dfrac{s}{s^2+\omega^2}, & \Re s > 0 \\[6pt]
\textbf{(6)} & \L[\sinh\omega t] = \dfrac{\omega}{s^2-\omega^2}, & \Re s > \abs{\omega} \\[6pt]
\textbf{(7)} & \L[\cosh\omega t] = \dfrac{s}{s^2-\omega^2}, & \Re s > \abs{\omega} \\[6pt]
\textbf{(8)} & \L[t\e^{at}] = \dfrac{1}{(s-a)^2}, & \Re s > \Re a \\[6pt]
\textbf{(9)} & \L[t\sin\omega t] = \dfrac{2\omega s}{(s^2+\omega^2)^2}, & \Re s > 0 \\[6pt]
\textbf{(10)} & \L[t\cos\omega t] = \dfrac{s^2-\omega^2}{(s^2+\omega^2)^2}, & \Re s > 0
\end{array}
\]
\end{theorem}

\begin{theorem}[单位脉冲与单位阶跃]
设 $\delta(t)$ 为单位脉冲函数，$u(t)$ 为单位阶跃函数．则
\[
\L[\delta(t)] = 1,\qquad \L[u(t)] = \frac{1}{s},\quad \Re s > 0.
\]
\end{theorem}

\begin{theorem}[周期函数的 Laplace 变换]
设 $f$ 在 $[0,+\infty)$ 上分段连续，周期为 $T$．则
\[
\L[f](s) = \frac{1}{1-\e^{-sT}}\int_0^{T} f(t)\e^{-st}\,\d t,\quad \Re s > 0.
\]
\end{theorem}

\begin{example}
    求下列函数的 Laplace 变换：
    \begin{tasks}[label=(\arabic*)](1)
        \task 以 $2\pi$ 为周期的函数 $f(t) = \begin{cases}
            \sin t, & 0\le t < \pi\\
            0, & \pi \le t < 2\pi
        \end{cases}$
        \task $f(t) = \dfrac{\sin at}{t},\quad a > 0$
    \end{tasks}
\end{example}
\begin{solution}
    \begin{enumerate}[label=(\arabic*)]
        \item 利用周期函数的 Laplace 变换公式，周期 $T=2\pi$：
        \[
        F(s) = \frac{1}{1-\e^{-2\pi s}}\int_0^{2\pi} f(t)\e^{-st}\,\d t
        = \frac{1}{1-\e^{-2\pi s}}\int_0^{\pi} \sin t\,\e^{-st}\,\d t.
        \]
        计算 $\displaystyle\int_0^{\pi} \sin t\,\e^{-st}\,\d t$：
        \[
        \int \sin t\,\e^{-st}\,\d t = -\frac{\e^{-st}(s\sin t+\cos t)}{s^2+1},
        \]
        故
        \[
        \int_0^{\pi} \sin t\,\e^{-st}\,\d t
        = \left[-\frac{\e^{-st}(s\sin t+\cos t)}{s^2+1}\right]_0^{\pi}
        = \frac{1+\e^{-\pi s}}{s^2+1}.
        \]
        代入得
        \[
        F(s) = \frac{1+\e^{-\pi s}}{(1-\e^{-2\pi s})(s^2+1)}
        = \frac{1}{(1-\e^{-\pi s})(s^2+1)},\qquad \Re s>0.
        \]

        \item 利用积分性质 $\displaystyle\L\!\left[\frac{g(t)}{t}\right]=\int_s^{+\infty}G(\sigma)\,\d\sigma$，其中 $g(t)=\sin at$，$G(s)=\dfrac{a}{s^2+a^2}$：
        \[
        F(s) = \int_s^{+\infty} \frac{a}{\sigma^2+a^2}\,\d\sigma
        = \left[\arctan\frac{\sigma}{a}\right]_s^{+\infty}
        = \frac{\pi}{2}-\arctan\frac{s}{a}
        = \arctan\frac{a}{s},\qquad \Re s>0.
        \]
    \end{enumerate}
\end{solution}

\section{Laplace逆变换的求法}

\subsection{部分分式法}

\begin{theorem}[部分分式展开]
若 $F(s)$ 为有理函数 $P(s)/Q(s)$，$\deg P < \deg Q$，且 $Q$ 有 $n$ 个互不相同的根 $s_1,\dots,s_n$，则
\[
F(s) = \sum_{k=1}^{n}\frac{c_k}{s-s_k},\qquad c_k = \Res(F,s_k) = \frac{P(s_k)}{Q'(s_k)},
\]
\[
f(t) = \L^{-1}[F] = \sum_{k=1}^{n} c_k \e^{s_k t}.
\]
\end{theorem}

\begin{theorem}[重根情形]
若 $s_0$ 为 $Q$ 的 $m$ 阶根，则
\[
\L^{-1}\!\left[\frac{1}{(s-s_0)^m}\right] = \frac{t^{m-1}}{(m-1)!}\e^{s_0 t}.
\]
\end{theorem}

\begin{example}
    求下列函数的 Laplace 逆变换：
    \begin{tasks}[label=(\arabic*)](2)
        \task $\displaystyle F(s) = \frac{1}{(s+1)^4}$
        \task $\displaystyle F(s) = \frac{(s+1)e^{-\pi s}}{s^2+s+1}$
    \end{tasks}
\end{example}
\begin{solution}
    \begin{enumerate}[label=(\arabic*)]
        \item 利用重根情形公式 $\displaystyle\L^{-1}\!\left[\frac{1}{(s-s_0)^m}\right]=\frac{t^{m-1}}{(m-1)!}\e^{s_0 t}$，其中 $s_0=-1$，$m=4$：
        \[
        f(t) = \frac{t^{3}}{3!}\e^{-t} = \frac{t^3}{6}\e^{-t},\quad t\ge 0.
        \]

        \item 先对不含延迟因子的部分求逆变换．令
        \[
        G(s)=\frac{s+1}{s^2+s+1}=\frac{s+\frac12}{\left(s+\frac12\right)^2+\left(\frac{\sqrt3}{2}\right)^2}
        +\frac{\frac12}{\left(s+\frac12\right)^2+\left(\frac{\sqrt3}{2}\right)^2}
        \]
        利用 $s$ 域平移性质（$a=-\frac12,\ \omega=\frac{\sqrt3}{2}$）：
        \[
        g(t)=\L^{-1}[G(s)]
        =\e^{-\frac{t}{2}}\cos\frac{\sqrt3}{2}t+\frac{1}{\sqrt3}\e^{-\frac{t}{2}}\sin\frac{\sqrt3}{2}t
        \]
        再利用 $t$ 域平移性质 $\L^{-1}[\e^{-\pi s}G(s)]=g(t-\pi)u(t-\pi)$，得
        \[
        f(t)=\e^{-\frac{t-\pi}{2}}\left[\cos\frac{\sqrt3}{2}(t-\pi)+\frac{1}{\sqrt3}\sin\frac{\sqrt3}{2}(t-\pi)\right]u(t-\pi)
        \]
    \end{enumerate}
\end{solution}

\subsection{卷积法}
\begin{theorem}[卷积法求逆变换]
若 $F(s)=F_1(s)F_2(s)$，且 $f_1=\L^{-1}[F_1]$、$f_2=\L^{-1}[F_2]$ 已知，则
\[
f(t) = \L^{-1}[F] = (f_1*f_2)(t) = \int_0^t f_1(\tau)f_2(t-\tau)\,\d\tau.
\]
\end{theorem}

\begin{example}
    求 $F(s) = \dfrac{1}{(s^2+4)^2}$ 的 Laplace 逆变换．
\end{example}
\begin{solution}
    利用卷积法：$\dfrac{1}{(s^2+4)^2}=\dfrac{1}{s^2+4}\cdot\dfrac{1}{s^2+4}$，而
    $\mathcal{L}^{-1}\!\left[\dfrac{1}{s^2+4}\right]=\dfrac{1}{2}\sin 2t$．故
    \[
    f(t)=\left(\frac{1}{2}\sin 2t\right)*\left(\frac{1}{2}\sin 2t\right)
    =\frac{1}{4}\int_0^t \sin(2\tau)\sin\bigl(2(t-\tau)\bigr)\,\d\tau.
    \]
    利用积化和差公式 $\sin A\sin B=\dfrac{1}{2}[\cos(A-B)-\cos(A+B)]$，其中
    $A=2\tau,\ B=2(t-\tau)$，得
    \begin{align*}
        f(t)&=\frac{1}{8}\int_0^t\left[\cos(4\tau-2t)-\cos(2t)\right]\,\d\tau
        =\frac{1}{8}\left[\frac{\sin(4\tau-2t)}{4}-\tau\cos(2t)\right]_0^t\\[1em]
        &=\frac{\sin 2t-2t\cos 2t}{16},\qquad t\ge 0
    \end{align*}

\end{solution}

\subsection{留数法}
\begin{theorem}[Bromwich 积分]
\[
f(t) = \L^{-1}[F](t) = \frac{1}{2\pi i}\int_{\sigma-i\infty}^{\sigma+i\infty} F(s)\e^{st}\,\d s = \sum_{k=1}^{n}\Res(F(s)\e^{st},s_k),
\]
其中 $s_k$ 为 $F(s)$ 的所有奇点，$\sigma$ 大于所有奇点的实部．
\end{theorem}

\begin{example}
求 $F(s)=\dfrac{s+3}{(s+1)(s-2)}$ 的 Laplace 逆变换．
\end{example}
\begin{solution}
部分分式分解：
\[
F(s) = \frac{A}{s+1} + \frac{B}{s-2},
\]
其中 $A = \dfrac{(-1)+3}{(-1)-2} = -\dfrac{2}{3}$，$B = \dfrac{2+3}{2+1} = \dfrac{5}{3}$．故
\[
f(t) = -\frac{2}{3}\e^{-t} + \frac{5}{3}\e^{2t},\quad t\ge 0
\]
\end{solution}

\begin{example}
求 $F(s)=\dfrac{1}{s(s^2+4)}$ 的 Laplace 逆变换．
\end{example}
\begin{solution}
部分分式：
\[
F(s) = \frac{1}{s(s^2+4)} = \frac{1}{4s} - \frac{s}{4(s^2+4)}
\]
故
\[
f(t) = \frac{1}{4} - \frac{1}{4}\cos 2t = \frac{1}{4}(1-\cos 2t)
\]
\end{solution}

\section{Laplace变换的应用}

Laplace 变换可以将微积分方程转化为代数方程，解出代数方程后再取逆变换即可得到微积分方程的解．

\subsection{常系数线性微分方程的初值问题}

\begin{example}
求初值问题 $y'' - 3y' + 2y = 4\e^{3t}$，$y(0)=1$，$y'(0)=-1$．
\end{example}
\begin{solution}
取 Laplace 变换：
\[
s^2 Y - s + 1 - 3(sY - 1) + 2Y = \frac{4}{s-3}
\]
整理得
\[
(s^2-3s+2)Y = \frac{4}{s-3} + s - 1 - 3 = \frac{4}{s-3} + s - 4
\]
\[
Y = \frac{4}{(s-1)(s-2)(s-3)} + \frac{s-4}{(s-1)(s-2)}
\]
部分分式后逐项逆变换得
\[
y(t) = 2\e^{3t} - \e^{2t}
\]
\end{solution}

\subsection{常系数线性微分方程组的求解}

\begin{example}
解方程组
\[
\begin{cases}
    x'+x-y=e^t,\\
    3x+y'-2y=2e^t
\end{cases},\quad x(0)=y(0)=1
\]
\end{example}
\begin{solution}
    设 $X(s)=\L[x(t)]$，$Y(s)=\L[y(t)]$．对方程组取 Laplace 变换，利用 $\L[e^t]=\dfrac{1}{s-1}$ 与初值 $x(0)=y(0)=1$：

    第一个方程：
    \[
    sX-1+X-Y=\frac{1}{s-1}
    \quad\Longrightarrow\quad
    (s+1)X-Y=\frac{s}{s-1} \tag{1}
    \]

    第二个方程：
    \[
    3X+sY-1-2Y=\frac{2}{s-1}
    \quad\Longrightarrow\quad
    3X+(s-2)Y=\frac{s+1}{s-1} \tag{2}
    \]

    由 (1) 得 $Y=(s+1)X-\dfrac{s}{s-1}$，代入 (2)：
    \[
    3X+(s-2)\left[(s+1)X-\frac{s}{s-1}\right]=\frac{s+1}{s-1}.
    \]

    整理得
    \[
    X(s)=\frac{1}{s-1}
    \]

    代回得
    \[
    Y(s)=(s+1)\cdot\frac{1}{s-1}-\frac{s}{s-1}=\frac{1}{s-1}.
    \]

    逆变换得
    \[
    x(t)=e^t,\qquad y(t)=e^t,\qquad t\ge 0
    \]
\end{solution}

\subsection{积分方程与微积分方程}

\begin{example}
求积分方程 $y(t) = \sin t + \displaystyle\int_0^t y(\tau)\,\d\tau$．
\end{example}
\begin{solution}
取 Laplace 变换：
\[
Y(s) = \frac{1}{s^2+1} + \frac{Y(s)}{s}
\]
解得
\[
Y(s) = \frac{s}{(s^2+1)(s-1)} = \frac{1}{2}\cdot \frac{1}{s-1} - \frac{1}{2}\cdot \frac{s}{s^2+1} + \frac{1}{2}\cdot \frac{1}{s^2+1}.
\]
故
\[
y(t) = \frac{1}{2}\e^{t} - \frac{1}{2}\cos t + \frac{1}{2}\sin t
\]
\end{solution}

\begin{example}
    求微积分方程 $\displaystyle y'(t)-4y(t) +4\int_0^t y(\tau)\,\d\tau = \frac{t^3}{3},\quad y(0)=0$．
\end{example}
\begin{solution}
    设 $Y(s)=\L[y(t)]$．对方程取 Laplace 变换得：
    \[
    sY-4Y+\frac{4}{s}Y=\frac{2}{s^4}
    \]
    整理得
    \[
    Y(s)\cdot\frac{s^2-4s+4}{s}=\frac{2}{s^4}
    \quad\Longrightarrow\quad
    Y(s)=\frac{2}{s^3(s-2)^2}
    \]
    作部分分式分解：
    \[
    \frac{2}{s^3(s-2)^2}
    =\frac{3/8}{s}+\frac{1/2}{s^2}+\frac{1/2}{s^3}-\frac{3/8}{s-2}+\frac{1/4}{(s-2)^2}
    \]
    逆变换得
    \[
    y(t)=\frac{3}{8}+\frac{t}{2}+\frac{t^2}{4}-\frac{3}{8}\e^{2t}+\frac{t}{4}\e^{2t}
    =\frac{3}{8}+\frac{t}{2}+\frac{t^2}{4}+\left(\frac{t}{4}-\frac{3}{8}\right)\e^{2t},\qquad t\ge 0
    \]
\end{solution}
